Smoke CPU thật, PowerShell đặt DISABLE_MODEL_SOURCE_CHECK=True rồi truyền script Python sau qua
`uv run --group ocr python -`. Flag chỉ bỏ bước dò kết nối host, models đã tải và dùng cache;
không bỏ kiểm TLS hay quyền nguồn.

```python
import importlib.metadata
import sys
from pathlib import Path
from PIL import Image, ImageDraw, ImageFont
from vireceipt.ocr import read_text
sys.stdout.reconfigure(encoding='utf-8')
folder = Path('runs/ocr-smoke')
folder.mkdir(parents=True, exist_ok=True)
image = Image.new('RGB', (1000, 400), 'white')
draw = ImageDraw.Draw(image)
font = ImageFont.truetype('C:/Windows/Fonts/arial.ttf', 44)
for index, line in enumerate(['CỬA HÀNG MINH ANH', 'Sữa tươi 2 x 32000 = 64000', 'TỔNG: 64000 VND']):
    draw.text((40, 40 + index * 100), line, fill='black', font=font)
path = folder / 'synthetic.png'
image.save(path)
for name in ['paddleocr', 'paddlepaddle', 'paddlex']:
    print(name, importlib.metadata.version(name), flush=True)
print('SMOKE_INPUT', path.as_posix(), flush=True)
print('OCR_TEXT_BEGIN', flush=True)
print(read_text(path), flush=True)
print('OCR_TEXT_END', flush=True)
```
