Smoke dùng ảnh tổng hợp đã tạo trong T2-06. Chạy script sau qua `uv run --group ocr python -`:

```text
import json
import os
import sys
import threading
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path
from vireceipt.extraction import extract_images, write_predictions
from vireceipt.extractors.ocr_llm import extract
from vireceipt.eval.io import load_predictions
sys.stdout.reconfigure(encoding='utf-8')
os.environ['DISABLE_MODEL_SOURCE_CHECK'] = 'True'
requests = []
class Handler(BaseHTTPRequestHandler):
    def do_POST(self):
        requests.append(json.loads(self.rfile.read(int(self.headers['Content-Length']))))
        body = json.dumps({'choices': [{'finish_reason': 'stop', 'message': {'content': '{"total":64000}', 'refusal': None}}]}).encode()
        self.send_response(200)
        self.send_header('Content-Length', str(len(body)))
        self.end_headers()
        self.wfile.write(body)
    def log_message(self, *args):
        pass
with ThreadingHTTPServer(('127.0.0.1', 0), Handler) as server:
    thread = threading.Thread(target=lambda: server.serve_forever(poll_interval=0.01), daemon=True)
    thread.start()
    os.environ.update({'VIRECEIPT_LLM_BASE_URL': f'http://127.0.0.1:{server.server_port}/v1', 'VIRECEIPT_LLM_MODEL': 'synthetic-text-stub', 'VIRECEIPT_LLM_API_KEY': 'synthetic-key'})
    try:
        predictions = extract_images({'r0001': Path('runs/ocr-smoke/synthetic.png')}, extract)
        output = Path('runs/ocr-llm-smoke/pred.jsonl')
        write_predictions(predictions, output)
        (prediction,) = load_predictions(output)
        assert len(requests) == 1 and prediction.receipt == {'total': 64000}
        assert prediction.decision == 'review' and prediction.extra['schema_valid'] is False
        print('LOCAL_OCR_TRANSCRIPT', requests[0]['messages'][1]['content'])
        print('PREDICTION', prediction.receipt, prediction.decision, prediction.extra)
        print('SMOKE_OK', output.as_posix())
    finally:
        server.shutdown()
        thread.join(timeout=2)
```
