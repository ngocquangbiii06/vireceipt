```text
import copy
import json
from datetime import date
from pathlib import Path
from vireceipt.extraction import ExtractionResult, extract_images
with open('tests/fixtures/gold.jsonl', encoding='utf-8') as stream:
    good=json.loads(next(stream))['receipt']
for scenario in ['mixed-large-float','sum-display-limit']:
    bad=copy.deepcopy(good)
    if scenario=='mixed-large-float':
        bad.update(items=[], subtotal=10**1000, total=10**1000, discount=1.0, vat=None)
    else:
        bad['items']=[{'name':'Synthetic','quantity':None,'unit_price':None,'amount':9*10**4299} for _ in range(2)]
        bad.update(subtotal=1,total=1)
    called=[]
    def backend(image):
        called.append(image.name)
        return ExtractionResult(bad if image.name=='a.png' else good)
    results=extract_images({'r0001':Path('a.png'),'r0002':Path('b.png')},backend,today=date(2026,10,23))
    assert called==['a.png','b.png'] and results[0].receipt==bad and results[1].receipt==good
    assert any(i['code']=='arithmetic_uncheckable' for i in results[0].extra['verification_issues'])
    print(scenario, 'both_images_preserved', called)
```
